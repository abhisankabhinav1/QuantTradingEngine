cmake_minimum_required(VERSION 3.20)
project(QuantTradingEngine VERSION 0.3.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
option(QTE_BUILD_TESTS "Build GoogleTest tests" ON)
option(QTE_BUILD_BENCHMARKS "Build benchmark executable" ON)
option(QTE_ENABLE_SANITIZERS "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)

add_library(qte STATIC src/engine.cpp)
target_include_directories(qte PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_features(qte PUBLIC cxx_std_20)
target_compile_options(qte PRIVATE $<$<CXX_COMPILER_ID:GNU,Clang>:-Wall;-Wextra;-Wpedantic>)
if(QTE_ENABLE_SANITIZERS AND CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(qte PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(qte PUBLIC -fsanitize=address,undefined)
endif()

add_executable(qte_demo src/main.cpp)
target_link_libraries(qte_demo PRIVATE qte)

if(QTE_BUILD_BENCHMARKS)
  add_executable(qte_benchmark benchmarks/benchmark_engine.cpp)
  target_link_libraries(qte_benchmark PRIVATE qte)
endif()

include(CTest)
if(BUILD_TESTING AND QTE_BUILD_TESTS)
  find_package(GTest QUIET)
  if(GTest_FOUND)
    add_executable(qte_tests
      tests/test_indicators.cpp tests/test_strategy.cpp tests/test_portfolio.cpp
      tests/test_execution.cpp tests/test_risk.cpp tests/test_backtest.cpp)
    target_link_libraries(qte_tests PRIVATE qte GTest::gtest_main)
    target_compile_options(qte_tests PRIVATE $<$<CXX_COMPILER_ID:GNU,Clang>:-Wall;-Wextra;-Wpedantic>)
    include(GoogleTest)
    gtest_discover_tests(qte_tests)
  else()
    message(STATUS "GoogleTest not found: qte_tests will not be built")
  endif()
endif()

install(TARGETS qte qte_demo ARCHIVE DESTINATION lib RUNTIME DESTINATION bin)
install(DIRECTORY include/ DESTINATION include)
