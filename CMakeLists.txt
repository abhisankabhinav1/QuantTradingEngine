cmake_minimum_required(VERSION 3.20)
project(QuantTradingEngine VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(QTE_BUILD_TESTS "Build GoogleTest tests" ON)
option(QTE_BUILD_BENCHMARKS "Build benchmark executable" ON)
add_library(qte STATIC src/engine.cpp)
target_include_directories(qte PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_options(qte PRIVATE $<$<CXX_COMPILER_ID:GNU,Clang>:-Wall;-Wextra;-Wpedantic>)
add_executable(qte_demo src/main.cpp)
target_link_libraries(qte_demo PRIVATE qte)
if(QTE_BUILD_BENCHMARKS)
 add_executable(qte_benchmark benchmarks/benchmark_engine.cpp)
 target_link_libraries(qte_benchmark PRIVATE qte)
endif()
if(QTE_BUILD_TESTS)
 include(CTest)
 enable_testing()
 find_package(GTest QUIET)
 if(GTest_FOUND)
  add_executable(qte_tests tests/test_indicators.cpp tests/test_strategy.cpp tests/test_portfolio.cpp tests/test_execution.cpp tests/test_risk.cpp tests/test_backtest.cpp)
  target_link_libraries(qte_tests PRIVATE qte GTest::gtest_main)
  include(GoogleTest)
  gtest_discover_tests(qte_tests)
 else()
  message(STATUS "GoogleTest not found; tests will not be built")
 endif()
endif()
